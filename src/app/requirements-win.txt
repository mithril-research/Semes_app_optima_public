# Windows 빌드 PC 에 설치할 패키지 (챔버 이상검사 콘솔 .exe 빌드용)
#   python -m venv venv && venv\Scripts\activate
#   python -m pip install -U pip
#   pip install -r requirements-win.txt
#
# ★★ 전 패키지를 '서로 맞는 2024-계열'로 상·하한 고정한다. 이유:
#   상한(<)이 없으면 재빌드 때 pip 가 최신(numpy 2.x, matplotlib 3.12, 최신 hooks-contrib)을
#   깔아버리는데, 이 최신들이 PyInstaller 번들링과 어긋나 부팅에러가 난다:
#     · numpy 2.x        → "cannot load module more than once per process"
#     · numpy 1.26 + '최신' hooks-contrib(numpy 2.x 전용) → "numpy.core._multiarray_tests" 류
#     · matplotlib 3.12  → "No module named _pyi_rth_mplconfig"
#   → numpy 1.26 과 '같은 세대'의 pyinstaller/hooks-contrib 로 통일해야 일관된다.
#   지원 파이썬: 3.9 ~ 3.12 (64-bit).

# ── 추론 엔진: onnxruntime (CPU) ★기본 ──
#   ★★ 기본은 반드시 CPU 판(onnxruntime)이다. GPU 판(onnxruntime-directml)은
#      DirectX 12 / DirectML.dll 을 요구해서, 지원이 약한 Win10 제어기에서는
#      네이티브 DLL 로드 단계에서 프로세스가 죽는다. 창 없는(windowed) exe 라
#      화면에 아무것도 안 뜨고 "더블클릭해도 반응 없음" 이 된다.
#      (Python 예외가 아니라 OS 로더 단계라 앱의 오류처리로도 못 잡는다)
#   GPU 가속이 필요하면 requirements-win-gpu.txt 로 빌드한다(그 PC 에서 확인된 경우만).
onnxruntime>=1.17.0,<1.23

# ── 추론/리포트 (numpy 1.x 세대로 고정) ──
numpy==1.26.4                  # ★ 고정. numpy 2.x 는 PyInstaller 중복로드 에러. 1.26 이 3.12 지원 최저 안전선.
pandas==2.2.2                  # ★ 정확 고정(결정적 빌드). numpy 1.26 호환.
matplotlib>=3.8,<3.10        # (exe 에선 제외됨) 리포트용. 3.10+ 는 PyInstaller 훅 미지원
# ★ scipy / scikit-learn / joblib 제거: exe 에 번들되지도 않고(spec 미수집) 런타임 미사용
#   (recon_explain._safe_auroc 는 sklearn 없으면 0.0 반환). scipy 는 설치 시 소스빌드로 자주
#   실패해 venv 를 불완전하게 만들고 "pandas dist-info 못 찾음" 같은 2차 빌드에러를 유발했음.

# ── 웹 서버 + API ──
fastapi>=0.110
uvicorn>=0.27
pydantic>=2.6

# ── WebView2 내장 빌드용 ──  (.NET Framework 4.7.2+ 필요)
pywebview==4.4.1
pythonnet==3.0.3
clr_loader>=0.2.6
cffi>=1.16.0

# ── 패키징 (numpy 1.26 과 '같은 세대'로 고정 — 최신으로 올리지 말 것) ──
#   최신 pyinstaller-hooks-contrib(2025+/2026)는 numpy 2.x 전용으로 맞춰져 numpy 1.26 과 어긋난다.
setuptools<81
wheel
pyinstaller>=6.10,<6.13
pyinstaller-hooks-contrib>=2024.8,<2025
